import { testDatabase } from './helpers/postgres';
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { analyzeIngredients } from '../src/lib/ingredients/analyze';
const now = 1800000000000;
const label = { rawText: 'Aqua, Retinol', complete: true, certain: true, confidence: 0.99, ingredients: [{ text: 'Aqua', confidence: 0.99 }, { text: 'Retinol', confidence: 0.99 }] };

test('ingredient drafts survive restart but are invalidated by photo revisions, catalog version and mode changes', async () => {
    const fixture = await testDatabase();
    let store = fixture.connect();
    try {
        const input = { name: 'Ada', email: 'cache@example.com', pregnancy: false, glp1: false };
        const start = await store.create(input, now);
        const token = await store.verify(start.token, start.code, now);
        await store.savePhotos(token, 0, [photo], now);
        const analysis = analyzeIngredients(label, { pregnancy: false, glp1: false });
        await store.saveIngredientAnalysis(token, 1, analysis, now);
        await store.close();
        store = fixture.connect();
        assert.equal((await store.ingredientAnalysis(token, 1, analysis.catalogVersion, now))?.ingredients.length, 2);
        assert.equal(await store.ingredientAnalysis(token, 1, 'different-version', now), null);
        const changed = await store.create({ ...input, pregnancy: true }, now + 60001);
        const newToken = await store.verify(changed.token, changed.code, now + 60001);
        assert.equal(await store.ingredientAnalysis(newToken, 1, analysis.catalogVersion, now + 60001), null);
        await assert.rejects(async () => await store.saveIngredientAnalysis(newToken, 1, analysis, now + 60001), /preferences changed/);
        await store.savePhotos(token, 1, [photo], now + 60001);
        assert.equal(await store.ingredientAnalysis(token, 2, analysis.catalogVersion, now + 60001), null);
        await assert.rejects(async () => await store.saveIngredientAnalysis(token, 1, analysis, now + 60001), /changed/);
        const next = await store.create(input, now + 86460002);
        const fresh = await store.verify(next.token, next.code, now + 86460002);
        await assert.rejects(async () => await store.ingredientAnalysis(fresh, 2, analysis.catalogVersion, now + 86460002), /expired/);
    }

        await fixture.cleanup();
    }
});

import { testDatabase } from './helpers/postgres';
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { NextRequest } from 'next/server';
import { intakeStore } from '../src/lib/intake';
import { POST } from '../src/app/api/ingredients/route';
import { CATALOG_VERSION } from '../src/lib/ingredients/catalog';
const label = ['Alcohol Denat.', 'Cetyl Alcohol', 'Retinol', 'Salicylic Acid 5%', 'Parfum'];
const transcript = () => ({ rawText: label.join(', '), complete: true, certain: true, confidence: 0.99, ingredients: label.map(text => ({ text, confidence: 0.99 })) });
const image = { bytes: Buffer.from('private previously quality-checked image'), width: 1000, height: 800 };
test('analysis uses only owned saved photos and server preferences; caches safely and never consumes a report', async () => {
    const fixture = await testDatabase();
    const before = { path: process.env.DATABASE_URL, key: process.env.LABEL_READER_API_KEY, fetch: globalThis.fetch };
    process.env.DATABASE_URL = fixture.url;
    process.env.LABEL_READER_API_KEY = 'test-placeholder';
    const store = intakeStore();
    const pending = await store.create({ name: 'Ada', email: 'ingredients@example.com', pregnancy: true, glp1: true });
    const token = await store.verify(pending.token, pending.code);
    const otherStart = await store.create({ name: 'Other', email: 'other-ingredients@example.com', pregnancy: false, glp1: false });
    const other = await store.verify(otherStart.token, otherStart.code);
    let calls = 0, lowConfidence = false, race = false;
    globalThis.fetch = async (_url, options) => {
        calls++;
        const request = JSON.parse(String(options?.body));
        assert.equal(request.response_format.json_schema.name, 'ingredient_transcription');
        if (race && calls % 2 === 0) {
            const current = await store.photoBatch(token);
            await store.savePhotos(token, current.revision, [image]);
        }
        return Response.json({ choices: [{ finish_reason: 'stop', message: { content: JSON.stringify({ ...transcript(), confidence: lowConfidence ? 0.5 : 0.99 }) } }] });
    };
    const request = (who: string, data: unknown = { revision: 1 }) => new NextRequest('http://localhost/api/ingredients', { method: 'POST', headers: { host: 'localhost', origin: 'http://localhost', 'content-type': 'application/json', cookie: `shelfie_session=${who}` }, body: JSON.stringify(data) });
    try {
        assert.equal((await POST(request(''))).status, 401);
        assert.equal((await POST(request(pending.token))).status, 401);
        assert.equal((await POST(request(token, { revision: 1, pregnancy: false }))).status, 400);
        assert.equal((await POST(request(token, { revision: 1, rawText: 'Glycerin' }))).status, 400);
        assert.equal((await POST(request(token))).status, 409);
        assert.equal(calls, 0);
        await store.savePhotos(token, 0, [image]);
        lowConfidence = true;
        assert.equal((await POST(request(token))).status, 422);
        assert.equal(await store.ingredientAnalysis(token, 1, CATALOG_VERSION), null);
        lowConfidence = false;
        const result = await POST(request(token));
        assert.equal(result.status, 200);
        const body = await result.json();
        assert.equal(body.status, 'complete');
        assert.deepEqual(body.modes, { pregnancy: true, glp1: true });
        assert.deepEqual(body.alcohols.drying, ['Alcohol Denat.']);
        assert.deepEqual(body.alcohols.fatty, ['Cetyl Alcohol']);
        assert.equal(body.ingredients[1].role, 'moisturizer');
        assert.equal(body.ingredients[1].flags.length, 0);
        assert.ok(body.ingredients[2].flags.some((flag: {
            kind: string;
        }) => flag.kind === 'pregnancy_nursing'));
        assert.ok(body.ingredients[4].flags.some((flag: {
            kind: string;
        }) => flag.kind === 'sensitive_skin'));
        const beforeCache = calls;
        assert.equal((await POST(request(token))).status, 200);
        assert.equal(calls, beforeCache);
        assert.equal((await POST(request(other))).status, 409);
        assert.equal((await store.requireEligible(token)).eligible, true);
        await store.savePhotos(token, 1, [image]);
        assert.equal(await store.ingredientAnalysis(token, 2, CATALOG_VERSION), null);
        const previousCalls = calls;
        race = true;
        // Make the next second read replace the photo batch to exercise the final transaction gate.
        calls = 0;
        assert.equal((await POST(request(token, { revision: 2 }))).status, 409);
        race = false;
        assert.ok(calls > 0);
        calls = previousCalls + calls;
        assert.equal((await store.requireEligible(token)).eligible, true);
        await store.completeReport(token, 'done-report', { products: [{}] });
        assert.equal((await POST(request(token, { revision: 3 }))).status, 409);
    }
    finally {
        globalThis.fetch = before.fetch;
        await store.close();
        await fixture.cleanup();
        for (const [key, value] of Object.entries({ DATABASE_URL: before.path, LABEL_READER_API_KEY: before.key })) {
            if (value === undefined)
                delete process.env[key];
            else
                process.env[key] = value;
        }
    }
});

import { test } from 'node:test';
import assert from 'node:assert/strict';
import { analyzeIngredients } from '../src/lib/ingredients/analyze';
import { catalog, findIngredient } from '../src/lib/ingredients/catalog';
import { parseIngredientList, validateTranscript } from '../src/lib/ingredients/parser';

const modes = { pregnancy: false, glp1: false };
function transcript(rawText: string) {
  return { rawText, complete: true, certain: true, confidence: 0.99, ingredients: parseIngredientList(rawText).map(item => ({ text: item.raw, confidence: 0.99 })) };
}

test('preserves every ingredient, punctuation within names, aliases, and printed percentages', () => {
  const parsed = parseIngredientList('Ingredients: Aqua (Water), 1,2-Hexanediol, Salicylic Acid (2%), Sodium Chloride.');
  assert.equal(parsed.length, 4); assert.equal(parsed[1].name, '1,2-Hexanediol'); assert.equal(parsed[2].concentrationPercent, 2);
  assert.equal(findIngredient(parsed[0].name)?.name, 'Aqua');
  assert.equal(parseIngredientList('Active Ingredients: Salicylic Acid 3%\nInactive Ingredients: Aqua, Glycerin').length, 3);
  assert.equal(parseIngredientList('2% Salicylic Acid, Aqua')[0].concentrationPercent, 2);
  assert.equal(parseIngredientList('Glycerin, Glyceryl Stearate (and) PEG-100 Stearate').length, 2);
});
test('rejects unreadable, incomplete, malformed or unverified lists without fixing them', () => {
  for (const input of ['', 'Aqua,,Glycerin', 'Aqua,', 'Aqua (Water, Glycerin', 'Aqua [Water)', 'Aqua, ???', 'Aqua, Retin...', 'Aqua, unreadable', 'Aqua, Salicylic Acid 200%', 'Aqua, Salicylic Acid (2% 3%)', 'Active Ingredients: Retinol', 'Aqua\nDirections: use daily']) assert.throws(() => parseIngredientList(input), input);
  const good = transcript('Aqua, Glycerin');
  for (const bad of [null, {}, { ...good, complete: false }, { ...good, certain: false }, { ...good, confidence: 0.97 }, { ...good, confidence: NaN }, { ...good, ingredients: [{text:'Aqua',confidence:0.99}] }, { ...good, ingredients: [{text:'Aqua',confidence:0.99},{text:'Retinol',confidence:0.99}] }, { ...good, ingredients: [{text:'Aqua',confidence:0.99},{text:'Glycerin',confidence:0.9}] }]) assert.throws(() => validateTranscript(bad));
  const printedPeriod = transcript('Aqua, Glycerin.');
  assert.equal(validateTranscript(printedPeriod).ingredients[1].name, 'Glycerin');
  assert.throws(() => validateTranscript({ ...printedPeriod, ingredients: good.ingredients }), /reliably/);
});
test('drying and fatty alcohol buckets are disjoint and aromatic/preservative names do not get substring matches', () => {
  const result = analyzeIngredients(transcript('Ethanol, Isopropyl Alcohol, Alcohol Denat., Cetyl Alcohol, Cetearyl Alcohol, Stearyl Alcohol, Benzyl Alcohol, Phenoxyethanol'), modes);
  assert.deepEqual(result.alcohols.drying, ['Ethanol', 'Isopropyl Alcohol', 'Alcohol Denat.']);
  assert.deepEqual(result.alcohols.fatty, ['Cetyl Alcohol', 'Cetearyl Alcohol', 'Stearyl Alcohol']);
  for (const item of result.ingredients.filter(item => item.alcoholBucket === 'fatty')) { assert.equal(item.role, 'moisturizer'); assert.equal(item.flags.length, 0); }
  for (const item of result.ingredients.filter(item => item.alcoholBucket === 'drying')) assert.equal(item.flags[0].kind, 'drying');
  assert.equal(result.ingredients[6].alcoholBucket, null); assert.equal(result.ingredients[7].alcoholBucket, null);
  assert.equal(findIngredient('Super Retinol Complex'), null); assert.equal(findIngredient('CetyI Alcohol'), null);
});
test('pregnancy and nursing flags use exact names and literal salicylic strength without blanket judgements', () => {
  const raw = 'Retinol, Retinaldehyde, Retinyl Palmitate, Hydroquinone, Salicylic Acid 5%, Salicylic Acid 2%, Salicylic Acid, Bakuchiol, Cetyl Alcohol';
  const off = analyzeIngredients(transcript(raw), modes);
  assert.equal(off.ingredients.flatMap(item => item.flags).filter(flag => flag.kind === 'pregnancy_nursing').length, 0);
  const on = analyzeIngredients(transcript(raw), { pregnancy: true, glp1: false });
  assert.ok(on.ingredients[4].flags.some(flag => flag.message.includes('above 2%')));
  assert.equal(on.ingredients[5].flags.length, 0);
  assert.ok(on.ingredients[6].flags.some(flag => flag.message.includes('strength is not printed')));
  assert.equal(on.ingredients[7].flags.length, 0); assert.equal(on.ingredients[8].flags.length, 0);
  for (const flag of on.ingredients.flatMap(item => item.flags)) assert.match(flag.message, /commonly flagged, check with your doctor/i);
  assert.doesNotMatch(JSON.stringify(on), /\b(?:safe|unsafe)\b/i);
});
test('GLP-1 mode adds gentle sensitivity flags to fragrance, acids and retinoids only when enabled', () => {
  const raw = 'Parfum, Glycolic Acid, Retinol, Cetyl Alcohol, Citric Acid';
  assert.equal(analyzeIngredients(transcript(raw), modes).ingredients.flatMap(item => item.flags).length, 0);
  const on = analyzeIngredients(transcript(raw), { pregnancy: false, glp1: true });
  assert.equal(on.ingredients.filter(item => item.flags.some(flag => flag.kind === 'sensitive_skin')).length, 3);
  assert.equal(on.ingredients[3].flags.length, 0); assert.equal(on.ingredients[4].flags.length, 0);
  assert.ok(on.notes.some(note => note.includes('does not infer a GLP-1 medication interaction')));
});
test('each ingredient has a plain-language line and unknown entries are explicit incomplete knowledge', () => {
  const result = analyzeIngredients(transcript('Aqua, Unlisted Botanical Extract, CetyI Alcohol'), modes);
  assert.equal(result.status, 'needs_review'); assert.equal(result.ingredients.length, 3);
  assert.equal(result.ingredients[1].match, 'unmatched'); assert.equal(result.ingredients[2].match, 'unmatched');
  assert.deepEqual(result.alcohols.fatty, []);
  assert.match(result.ingredients[1].explanation, /unknown/);
  for (const entry of catalog) {
    assert.ok(entry.explanation.length > 10); assert.ok(!entry.explanation.includes('\n'));
    assert.doesNotMatch(entry.explanation, /\b(?:humectant|emollient|surfactant|keratolytic|diagnose|treat|cure|guarantee|safe|unsafe)\b/i);
  }
  assert.equal(analyzeIngredients(transcript('Aqua, Glycerin'), modes).status, 'complete');
});

import { test } from 'node:test';
import assert from 'node:assert/strict';
import sharp from 'sharp';
import { checkLabelQuality, enforceVerdict, preparePhoto } from '../src/lib/photo-quality';

export const labelSvg = `<svg width="1200" height="1000" xmlns="http://www.w3.org/2000/svg"><rect width="1200" height="1000" fill="white"/><g fill="black" font-family="sans-serif" font-size="44"><text x="90" y="140">SHELFIE TEST CLEANSER</text><text x="90" y="260">INGREDIENTS:</text><text x="90" y="380">Aqua, Glycerin, Cetyl Alcohol,</text><text x="90" y="480">Stearyl Alcohol, Sodium Chloride,</text><text x="90" y="580">Citric Acid, Phenoxyethanol.</text><text x="90" y="800">END OF INGREDIENT LIST</text></g></svg>`;
const verdict = (count = 1) => ({ sameProduct: true, completeIngredientList: true, certain: true, photos: Array.from({ length: count }, (_, index) => ({ index, blurry: false, dark: false, croppedText: false, glare: false, readable: true, confidence: 0.99 })) });

test('decodes a detailed image, strips metadata and rejects real dark, blurred and invalid inputs', async () => {
  const source = await sharp(Buffer.from(labelSvg)).png().toBuffer();
  const result = await preparePhoto(source);
  assert.equal((await sharp(result.bytes).metadata()).format, 'jpeg');
  assert.equal((await sharp(result.bytes).metadata()).exif, undefined);
  const dark = await sharp(source).modulate({ brightness: 0.04 }).png().toBuffer();
  await assert.rejects(preparePhoto(dark), /dark/);
  const blurry = await sharp(source).blur(35).png().toBuffer();
  await assert.rejects(preparePhoto(blurry), /blurry/);
  await assert.rejects(preparePhoto(Buffer.from('not an image')), /couldn’t open/);
  await assert.rejects(preparePhoto(await sharp(source).resize(200).png().toBuffer()), /resolution/);
  await assert.rejects(preparePhoto(Buffer.alloc(6 * 1024 * 1024 + 1)), /6 MB/);
});

test('fail-closed verdict handles blur, darkness, cropping, glare, confidence and incomplete multi-photo coverage', () => {
  for (const field of ['blurry', 'dark', 'croppedText', 'glare'] as const) {

    const fixture = await testDatabase();

    const other = await store.create({ ...input, email: 'other-route@example.com' });
    const otherToken = await store.verify(other.token, other.code);
    const image = await sharp(Buffer.from(svg)).png().toBuffer();
    let verdict: Record<string, unknown> = {};
    let calls = 0;
    globalThis.fetch = async (_url, options) => {
        calls++;
        const body = JSON.parse(String(options?.body));
        const count = body.messages[1].content.length - 1;
        return Response.json({ choices: [{ finish_reason: 'stop', message: { content: JSON.stringify({ sameProduct: true, completeIngredientList: true, certain: true,
                            photos: Array.from({ length: count }, (_, index) => ({ index, blurry: false, dark: false, croppedText: false, glare: false, readable: true, confidence: 0.99, ...verdict })) }) } }] });
    };
    const request = (cookie: string, files: Buffer[] = [image], existingIds: string[] = [], revision = 0) => {
        const body = new FormData();
        body.set('revision', String(revision));
        body.set('existingIds', JSON.stringify(existingIds));
        for (const file of files)
            body.append('photos', new Blob([new Uint8Array(file)], { type: 'image/png' }), 'label.png');

    };

        assert.equal((await POST(request(''))).status, 401);
        assert.equal((await POST(request(pending.token))).status, 401);

        verdict = { croppedText: true };
        assert.equal((await POST(request(token))).status, 422);
        assert.equal((await store.photoBatch(token)).photos.length, 0);
