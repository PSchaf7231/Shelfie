import { cookies } from "next/headers";
import { redirect } from "next/navigation";
import ReportContent from "@/components/report-content";
import ReportScreen from "@/components/report-screen";
import { intakeStore, IntakeError } from "@/lib/intake";
import { SESSION_COOKIE } from "@/lib/intake-http";
import { CATALOG_VERSION } from "@/lib/ingredients/catalog";
import { REPORT_DISCLAIMER } from "@/lib/report";

export const runtime = "nodejs";
export default async function ReportPage() {
  const token = (await cookies()).get(SESSION_COOKIE)?.value || "";
  const store = intakeStore();
  if ((await store.status(token)).state !== "verified") redirect("/intake");
  let content;
  try {
    const report = await store.getReport(token);
    if (report) content = <ReportContent report={report} />;
    else {
      const batch = await store.photoBatch(token);
      if (!batch.photos.length || !await store.ingredientAnalysis(token, batch.revision, CATALOG_VERSION)) redirect("/scan");
      content = <ReportScreen revision={batch.revision} />;
    }
  } catch (error) {
    if (!(error instanceof IntakeError)) throw error;
    if (error.status === 401) redirect("/intake");
    if (error.status === 409) redirect("/scan");
    content = <section className="intake-panel"><h1>We couldn’t open your report.</h1><p className="form-error" role="alert">{error.message}</p><a className="back-link" href="/report">Try opening it again</a></section>;
  }
  return <div className="intake-shell report-shell">
    <header><a className="brand brand-link" href="/">Shelfie</a></header>
    <main>{content}</main>
    <footer>{REPORT_DISCLAIMER}</footer>
  </div>;
}

"use client";
import { useEffect, useState, type FormEvent } from "react";
type Status = { state: "intake" | "pending" | "verified"; email?: string; eligible?: boolean; pregnancy?: boolean; glp1?: boolean };
export default function IntakeForm({ initialPregnancy }: { initialPregnancy: boolean }) {
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");
  const [pregnancy, setPregnancy] = useState(initialPregnancy);
  const [glp1, setGlp1] = useState(false);
  const [code, setCode] = useState("");
  const [status, setStatus] = useState<Status>({ state: "intake" });
  const [busy, setBusy] = useState(false);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  useEffect(() => {
    let active = true;
    fetch("/api/intake/status", { cache: "no-store" }).then(async response => {
      if (!response.ok) throw new Error("Couldn’t check your verification. Please refresh to try again.");
      return response.json();
    }).then(data => { if (active) { setStatus(data); if (data.email) setEmail(data.email); } })
      .catch(error => { if (active) setError(error.message); })
      .finally(() => { if (active) setLoading(false); });
    return () => { active = false; };
  }, []);
  async function submit(event: FormEvent, verify = false) {
    event.preventDefault(); setBusy(true); setError("");
    try {
      const response = await fetch(verify ? "/api/verify-email" : "/api/intake", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify(verify ? { code } : { name, email, pregnancy, glp1 }),
      });
      const data = await response.json();
      if (!response.ok) throw new Error(data.error || "Please try again.");
      setStatus(data); setCode("");
    } catch (error) { setError(error instanceof Error ? error.message : "Please try again."); }
    finally { setBusy(false); }
  }
  if (loading) return <p role="status">Checking your verification…</p>;
  if (status.state === "verified") return <section className="intake-panel">
    <h1>{status.eligible ? "You’re verified." : "Your free report is already used."}</h1>
    <p className="form-intro">{status.eligible ? "Your first report is still free. You can now photograph your product label." : "This email has already received its completed free report."}</p>
    <p className="form-help">Email verified: {status.email}</p>
    <ul className="preferences"><li>Pregnancy &amp; nursing mode: {status.pregnancy ? "On" : "Off"}</li><li>GLP-1 sensitive-skin mode: {status.glp1 ? "On" : "Off"}</li></ul>
    {status.eligible && <a href="/scan" className="primary-button form-button">Photograph my label</a>}
    {!status.eligible && <a href="/report" className="primary-button form-button">View my report</a>}
    <a href="/" className="back-link">Back to Shelfie</a>
  </section>;
  return <section className="intake-panel">
    <h1>{status.state === "pending" ? "Check your email." : "Your first report starts here."}</h1>
    <p className="form-intro">{status.state === "pending" ? `Enter the six-digit code sent to ${status.email}. It expires in 10 minutes.` : "Your first report is free — no credit card needed."}</p>
    {status.state === "intake" ? <form onSubmit={event => submit(event)}>
      <label className="field">Name<input name="name" autoComplete="given-name" required maxLength={100} value={name} onChange={event => setName(event.target.value)} /></label>
      <label className="field">Email<input name="email" type="email" autoComplete="email" required maxLength={254} value={email} onChange={event => setEmail(event.target.value)} /></label>
      <fieldset className="mode-options"><legend>Report preferences</legend>
        <label className="toggle-row"><span>Pregnancy &amp; nursing mode</span><input type="checkbox" role="switch" checked={pregnancy} onChange={event => setPregnancy(event.target.checked)} /></label>
        <label className="toggle-row"><span>GLP-1 sensitive-skin mode</span><input type="checkbox" role="switch" checked={glp1} onChange={event => setGlp1(event.target.checked)} /></label>
      </fieldset>
      <p className="form-help">Verify your email before scanning. One completed free report per email.</p>
      <button className="primary-button form-button" disabled={busy}>{busy ? "Sending your code…" : "Send verification code"}</button>
    </form> : <form onSubmit={event => submit(event, true)}>
      <label className="field">Verification code<input name="code" inputMode="numeric" autoComplete="one-time-code" pattern="[0-9]{6}" maxLength={6} required value={code} onChange={event => setCode(event.target.value.replace(/\D/g, ""))} /></label>
      <button className="primary-button form-button" disabled={busy}>{busy ? "Verifying…" : "Verify my email"}</button>
      <p className="form-help">Didn’t get a code? Check your spam folder. You can request a new code after one minute.</p>
      <button type="button" className="text-button" disabled={busy} onClick={() => { setStatus({ state: "intake" }); setError(""); setCode(""); }}>Request a new code or change email</button>
    </form>}
    {error && <p role="alert" className="form-error">{error}</p>}
  </section>;
}

"use client";
import { useEffect, useRef, useState, type ChangeEvent } from "react";

type Photo = { id: string; url: string; file?: File };
export default function PhotoCapture() {
  const camera = useRef<HTMLInputElement>(null);
  const upload = useRef<HTMLInputElement>(null);
  const urls = useRef(new Set<string>());
  const [photos, setPhotos] = useState<Photo[]>([]);
  const [revision, setRevision] = useState(0);
  const [busy, setBusy] = useState(false);
  const [reading, setReading] = useState(false);
  const [changed, setChanged] = useState(false);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [message, setMessage] = useState("");
  const [locked, setLocked] = useState(false);
  const working = busy || reading;

  useEffect(() => {
    let active = true;
    fetch("/api/photos", { cache: "no-store" }).then(async response => {
      const data = await response.json();
      if (!response.ok) {
        if (active && [401, 409].includes(response.status)) setLocked(true);
        throw new Error(data.error || "Couldn’t load your saved photos. Refresh to try again.");
      }
      if (active) { setPhotos(data.photos); setRevision(data.revision); if (data.photos.length) setMessage("Your label photos are saved. Your free report is still available."); }
    }).catch(error => { if (active) setError(error.message); })
      .finally(() => { if (active) setLoading(false); });
    const ownedUrls = urls.current;
    return () => { active = false; for (const url of ownedUrls) URL.revokeObjectURL(url); ownedUrls.clear(); };
  }, []);

  function select(event: ChangeEvent<HTMLInputElement>) {
    const files = Array.from(event.target.files || []);
    event.target.value = ""; // permits selecting the same file for a retake
    setError(""); setMessage("");
    if (files.length + photos.length > 6) { setError("Use up to six photos of one product. Remove a photo before adding more."); return; }
    if (files.some(file => file.size > 6 * 1024 * 1024)) { setError("Each photo must be 6 MB or smaller. Please choose a smaller photo."); return; }
    if (files.length) setChanged(true);
    setPhotos(current => [...current, ...files.map(file => { const url = URL.createObjectURL(file); urls.current.add(url); return { id: crypto.randomUUID(), url, file }; })]);
  }
  function remove(photo: Photo) {
    if (photo.file) { URL.revokeObjectURL(photo.url); urls.current.delete(photo.url); }
    setPhotos(current => current.filter(item => item.id !== photo.id)); setChanged(true); setMessage(""); setError("");
  }
  async function check() {
    setBusy(true); setError(""); setMessage("");
    try {
      const body = new FormData();
      body.set("revision", String(revision));
      body.set("existingIds", JSON.stringify(photos.filter(photo => !photo.file).map(photo => photo.id)));
      for (const photo of photos) if (photo.file) body.append("photos", photo.file);
      const response = await fetch("/api/photos", { method: "POST", body });
      const data = await response.json();
      if (!response.ok) {
        if (response.status === 401) setLocked(true);
        if (response.status === 409) setError((data.error || "Your photos changed.") + " Refresh to continue.");
        throw new Error(data.error || "Couldn’t check your photos. Please try again.");
      }
      for (const url of urls.current) URL.revokeObjectURL(url);
      urls.current.clear(); setPhotos(data.photos); setRevision(data.revision); setChanged(false); setMessage(data.message);
    } catch (error) { setError(error instanceof Error ? error.message : "Please try again."); }
    finally { setBusy(false); }
  }
  async function openReport() {
    setReading(true); setError("");
    try {
      const response = await fetch("/api/ingredients", {

"use client";
import { useEffect, useState } from "react";
import ReportContent from "./report-content";
import { isReportAnalysis, type CompletedReport } from "@/lib/report";

export default function ReportScreen({ revision }: { revision: number }) {
  const [report, setReport] = useState<CompletedReport | null>(null);
  const [error, setError] = useState("");
  const [retry, setRetry] = useState(0);
  const [checking, setChecking] = useState(true);
  useEffect(() => {
    let active = true;
    setChecking(true); setError("");
    // Completion is a same-origin POST when this screen opens. Page GETs and
    // prefetches never consume a report. The server makes repeats idempotent.
    fetch("/api/report", {
      method: "POST", headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ revision }),
    }).then(async response => {
      const data = await response.json();
      if (!response.ok) throw new Error(data.error || "We couldn’t open your report. Please try again.");
      if (typeof data.id !== "string" || !Number.isFinite(data.completed) || !isReportAnalysis(data.analysis)) throw new Error("We couldn’t open your saved report. Please try again.");
      if (active) setReport(data);
    }).catch(error => { if (active) setError(error instanceof Error ? error.message : "Please try again."); })
      .finally(() => { if (active) setChecking(false); });
    return () => { active = false; };
  }, [revision, retry]);
  if (report) return <ReportContent report={report} />;
  return <section className="intake-panel report-opening">
    <h1>{checking ? "Opening your report." : "Let’s try that again."}</h1>
    {checking && <p className="form-intro" role="status">Saving your completed report so you can come back to it.</p>}
    {error && <>
      <p className="form-error" role="alert">{error}</p>
      <button className="primary-button form-button" onClick={() => setRetry(current => current + 1)}>Open my report again</button>
      <a className="back-link" href="/scan">Back to label photos</a>
    </>}
  </section>;
}
